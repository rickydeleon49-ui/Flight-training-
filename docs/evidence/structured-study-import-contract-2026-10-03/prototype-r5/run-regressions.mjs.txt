// Ordinary ASCII private text that is also identifier-shaped, plus credential-shaped values built at run time.
const PRIVATE = ["PrivateNote_Alice42", "Alice.medical.2026", "acct-7741-checking"];
const CREDENTIAL_SHAPED = [
  ["sk-", "Q", 30], ["ghp_", "A", 36], ["AKIA", "B", 16], ["AIza", "C", 35], ["xoxb-", "1", 12],
].map(([prefix, ch, n]) => prefix + ch.repeat(n));
const SENSITIVE = [...PRIVATE, ...CREDENTIAL_SHAPED];
// Choice IDs and answer keys are limited to 32 characters, so only sentinels that are valid choice IDs are used there.
const CHOICE_SAFE = SENSITIVE.filter((v) => /^[A-Za-z0-9][A-Za-z0-9_.-]{0,31}$/.test(v));

const ALLOWED_FAILURE_KEYS = new Set(["ok", "stage", "errors", "omittedErrors"]);
const ALLOWED_ERROR_KEYS = new Set(["code", "path", "message", "limit", "offset", "supported", "index", "firstIndex", "field"]);
const FIELD_ALLOWLIST = new Set([
  // contract-owned names (duplicate_key on a known field)
  "format", "formatVersion", "title", "source", "generator", "createdAt", "note", "guide", "summary", "sections",
  "heading", "body", "sourceNotes", "flashcards", "id", "front", "back", "sourceRefs", "label", "page", "questions",
  "type", "question", "choices", "text", "correctChoiceId", "explanation",
  // fixed directive and legacy-answer allowlists
  "actions", "commands", "command", "script", "scripts", "exec", "execute", "run", "fetch", "upload", "download",
  "webhook", "callback", "instructions", "system", "systemPrompt", "onImport", "onLoad", "html", "mdx", "jsx",
  "apiKey", "token", "credentials", "correctIndex", "answerIndex", "answer", "correct", "correctAnswer",
  "correctChoice", "answers", "correctChoiceIds",
]);
const PATH_RE = /^\$(?:\[\d+\]|\.(?:<field>|[A-Za-z]+))*$/;

function leaks(result, values = SENSITIVE) {
  const text = JSON.stringify(result);
  return values.filter((v) => text.includes(v)).length;
}

/** Whole-failure-object invariant: allowlisted keys, fixed path grammar, allowlisted `field`, no sensitive values. */
function failureShapeProblems(result) {
  const problems = [];
  if (result.ok !== false) problems.push("not a failure");
  for (const k of Object.keys(result)) if (!ALLOWED_FAILURE_KEYS.has(k)) problems.push(`failure key ${k}`);
  for (const e of result.errors ?? []) {
    for (const k of Object.keys(e)) if (!ALLOWED_ERROR_KEYS.has(k)) problems.push(`error key ${k}`);
    if (typeof e.path !== "string" || !PATH_RE.test(e.path)) problems.push("path grammar");
    for (const seg of (e.path ?? "").match(/\.[A-Za-z]+/g) ?? []) if (!FIELD_ALLOWLIST.has(seg.slice(1))) problems.push("path segment not contract-owned");
    if ("field" in e && !FIELD_ALLOWLIST.has(e.field)) problems.push("field not allowlisted");
    for (const k of ["limit", "offset", "index", "firstIndex"]) if (k in e && !Number.isInteger(e[k])) problems.push(`${k} not integer`);
    if (typeof e.message !== "string") problems.push("message");
  }
  if (leaks(result)) problems.push(`sensitive values echoed: ${leaks(result)}`);
  return problems;
}

const rows = [];
function test(name, fn) {
  let problems;
  try { problems = fn() ?? []; } catch (e) { problems = [`threw ${e?.constructor?.name ?? "error"}`]; }
  rows.push({ name, pass: problems.length === 0, problems });
}
const codes = (r) => [...new Set((r.errors ?? []).map((e) => e.code))].sort().join(",");
const expectCode = (r, code) => (r.ok === false && (r.errors ?? []).some((e) => e.code === code) ? [] : [`expected ${code}, got ${r.ok ? "ok" : codes(r)}`]);

// ---------------------------------------------------------------- R1: diagnostic privacy
test("R1: unknown ASCII private keys are redacted at top level and nested", () => {
  const p = [];
  for (const key of PRIVATE) {
    for (const packet of [base({ guide: { summary: "x" }, [key]: "v" }), base({ guide: { summary: "x", [key]: "v" } }),
      base({ questions: [q("q1", { [key]: 1 })] })]) {
      const r = run(packet);
      p.push(...expectCode(r, "unknown_field"), ...failureShapeProblems(r));
      if ((r.errors ?? []).some((e) => e.code === "unknown_field" && "field" in e)) p.push("unknown_field carries field");
    }
  }
  return p;
});
test("R1: credential-shaped unknown keys are redacted", () => {
  const p = [];
  for (const key of CREDENTIAL_SHAPED) {
    const r = run(base({ flashcards: [card("c1", { [key]: true })], [key]: "v" }));
    p.push(...expectCode(r, "unknown_field"), ...failureShapeProblems(r));
  }
  return p;
});
test("R1: nested scan failures under unknown keys use redacted paths", () => {
  const p = [];
  for (const key of SENSITIVE) {
    const dup = validatePacketBytes(enc(`{"format":"hfd-study-packet","formatVersion":1,"title":"x","${key}":{"title":"a","title":"b"}}`));
    p.push(...expectCode(dup, "duplicate_key"), ...failureShapeProblems(dup));
    if (dup.errors?.[0]?.path !== "$.<field>") p.push("duplicate path not redacted");
    const deep = validatePacketBytes(enc(`{"title":"x","${key}":[[[[[[[[[0]]]]]]]]]}`));
    p.push(...expectCode(deep, "depth_exceeded"), ...failureShapeProblems(deep));
    const bad = validatePacketBytes(enc(`{"title":"x","${key}":{"${key}":[1,}}`));
    p.push(...expectCode(bad, "malformed_json"), ...failureShapeProblems(bad));
    const ctl = validatePacketBytes(enc(`{"title":"x","${key}":{"n":"a\\u0001b"}}`));
    p.push(...expectCode(ctl, "invalid_string_character"), ...failureShapeProblems(ctl));
  }
  return p;
});

  ["question", (id) => base({ questions: [q(id), q("other"), q(id)] }), "duplicate_id", "$.questions[2].id"],

  });
}

test("R1: failure objects contain only allowlisted properties across stages", () => {
  const p = [];
  const cases = [
    new Uint8Array(0), new Uint8Array([0xff, 0xfe, 0x7b]), enc("{"), enc('{"format":"x"}'),
    enc(JSON.stringify(base({ formatVersion: 7, guide: { summary: "x" } }))),
    enc(JSON.stringify(base({ title: 1 }))), enc(JSON.stringify(base())),
    enc(JSON.stringify(base({ questions: Array.from({ length: 60 }, (_, k) => q(`q${k}`, { correctChoiceId: "z" })) }))),
  ];
  for (const bytes of cases) p.push(...failureShapeProblems(validatePacketBytes(bytes)));
  const version = validatePacketBytes(cases[4]);
  if ("found" in (version.errors?.[0] ?? {})) p.push("unsupported_version returns the packet's version value");
  const capped = validatePacketBytes(cases[7]);
  if (capped.errors?.length !== 50 || capped.omittedErrors !== 10) p.push("error cap");
  return p;
});
test("R1: unexpected internal errors return a fixed code without exception text", () => {

  }
  return p;

test("R3: canonical persistence JSON ignores key order and preview-only values", () => {
  const { canonicalPersistenceJson } = validator;
  if (typeof canonicalPersistenceJson !== "function") return ["canonicalPersistenceJson missing"];
  const a = run(REF_PACKET);
  const reordered = JSON.parse(JSON.stringify(REF_PACKET), (k, v) => (v && typeof v === "object" && !Array.isArray(v) ? Object.fromEntries(Object.entries(v).reverse()) : v));
  const b = run({ ...reordered, source: { note: "Different claim" } });
  const c = run({ ...REF_PACKET, flashcards: REF_PACKET.flashcards.map((x, i) => ({ ...x, id: `renamed-${i}`, sourceRefs: [] })) });
  const d = run({ ...REF_PACKET, questions: [q("question-gamma-79", { correctChoiceId: "a" }), REF_PACKET.questions[1]] });
  const [ja, jb, jc, jd] = [a, b, c, d].map((r) => canonicalPersistenceJson(r.payload));
  const p = [];
  if (ja !== jb || ja !== jc) p.push("canonical JSON changed with key order, claims, IDs or refs");
  if (ja === jd) p.push("canonical JSON did not change with the answer");
  for (const v of ["card-alpha-77", "Invented chart", "Different claim"]) if (ja.includes(v) || jb.includes(v)) p.push("preview-only value in canonical JSON");
  return p;
});
test("R3: text entry re-derives the byte limit and refuses an unstripped BOM", () => {
  const { validatePacketText } = validator;
  if (typeof validatePacketText !== "function") return ["validatePacketText missing"];
  const small = JSON.stringify(base({ guide: { summary: "x" } }));
  const atLimitMinus3 = small + " ".repeat(4 * 1024 * 1024 - 3 - Buffer.byteLength(small));
  const p = [];
  if (!validatePacketText(atLimitMinus3).ok) p.push("rejected under limit");
  const withBom = validatePacketText(atLimitMinus3, { bomRemoved: true });
  if (!withBom.ok || !withBom.preview.warnings.some((w) => w.code === "utf8_bom_removed")) p.push("BOM at exact limit");
  const over = validatePacketText(atLimitMinus3 + " ", { bomRemoved: true });
  if (over.ok || over.errors[0].code !== "file_too_large") p.push("over limit with BOM accepted");
  const astral = base({ guide: { summary: "\u{1F680}".repeat(1_048_576) } }); // 4 UTF-8 bytes each, 2 UTF-16 units
  const big = validatePacketText(JSON.stringify(astral));
  if (big.ok || big.errors[0].code !== "file_too_large") p.push("UTF-8 length not derived from text");
  const bom = validatePacketText("\uFEFF" + small);
  if (bom.ok || bom.errors[0].code !== "unsupported_encoding") p.push("leading BOM character accepted");
  return p;
});

// ---------------------------------------------------------------- R5: guide source notes need a persistable study set

const rejectedCleanly = (r, wantCodes) => {
  const p = [...failureShapeProblems(r)];
  if ("preview" in r || "payload" in r) p.push("rejected result carries preview or payload");
  if (codes(r) !== wantCodes) p.push(`codes ${codes(r)}`);
  return p;
};
const orphanPath = (r) => (r.errors?.some((e) => e.code === "source_notes_without_study_set" && e.path === "$.guide.sourceNotes") ? [] : ["expected source_notes_without_study_set at $.guide.sourceNotes"]);

  const r = run(base({ questions: [q("q1")], guide: { sourceNotes: NOTE } }));

const keptNotes = (r, want) => {
  const p = [];
  if (!r.ok) return [`rejected ${codes(r)}`];
  if (!r.preview.createsStudySet) p.push("createsStudySet false");
  if (r.payload.studySet?.sourceNotes !== want) p.push("payload notes");
  if (r.preview.data.guide?.sourceNotes !== want) p.push("preview notes");
  if ("sourceNotes" in r.preview.discardedOnSave || JSON.stringify(r.preview.discardedOnSave).includes(want)) p.push("notes listed as discarded");
  return p;
};
