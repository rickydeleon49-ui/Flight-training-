// Disposable prototype of safe guide rendering with react-markdown 10.1.0.
// Contract §4. Raw HTML is shown as inert escaped text, MDX is never enabled (no third-party remark/rehype plugins),
// links are never navigable, images are never loaded. Not production code.
import { createElement as h } from "react";
import { renderToStaticMarkup } from "react-dom/server";
import Markdown from "react-markdown";

/** Elements the guide renderer may produce. Everything else is unwrapped to its text. */
export const ALLOWED_ELEMENTS = Object.freeze([
  "p", "h1", "h2", "h3", "h4", "h5", "h6", "ul", "ol", "li", "em", "strong", "code", "pre",
  "blockquote", "hr", "br", "a", "img",
]);

/** Deepest parsed Markdown tree (in nodes, root = 1) rendered with formatting. Proposed; a list level uses two nodes. */
export const MAX_MARKDOWN_TREE_DEPTH = 64;

const LOCAL_HOST = /^(localhost|.*\.localhost|127(?:\.\d{1,3}){3}|\[::1\]|0\.0\.0\.0)$/i;

/**
 * Returns the URL only when it is worth *showing as text*; nothing is ever navigated or fetched.
 * Only absolute http(s) URLs to non-local hosts and mailto: survive. Relative, protocol-relative, local hosts
 * (including asset.localhost and tauri.localhost), file:, asset:, tauri:, data:, javascript: and every other
 * scheme become "".
 */
export function displayUrl(url) {
  const trimmed = String(url ?? "").trim();
  if (/^mailto:[^\s]+$/i.test(trimmed)) return trimmed;
  if (!/^https?:\/\//i.test(trimmed)) return "";
  try {
    const parsed = new URL(trimmed);
    if (LOCAL_HOST.test(parsed.hostname) || parsed.username || parsed.password) return "";
    return trimmed;
  } catch {
    return "";
  }
}

/**
 * Our own remark transformer (not a third-party plugin). Measures the parsed tree iteratively and, when it is
 * deeper than the limit, replaces it with the original text as one paragraph. Parsing is iterative; the later
 * recursive conversion is what overflowed in the prototype.
 */
export function remarkDepthGuard(stats, source) {
  return () => (tree) => {
    let max = 0;
    const stack = [[tree, 1]];
    while (stack.length) {
      const [node, depth] = stack.pop();
      if (depth > max) max = depth;
      if (max > MAX_MARKDOWN_TREE_DEPTH) break;
      for (const child of node.children ?? []) stack.push([child, depth + 1]);
    }
    stats.treeDepth = max;
    if (max > MAX_MARKDOWN_TREE_DEPTH) {
      stats.plainTextFallback = true;
      tree.children = [{ type: "paragraph", children: [{ type: "text", value: source }] }];
    }
  };
}

const headingLevel = (tag) => (tag === "h1" || tag === "h2" ? "h5" : "h6");

export function guideComponents(stats) {
  const heading = (tag) => ({ children }) => h(headingLevel(tag), { className: "md-heading" }, children);
  return {
    h1: heading("h1"), h2: heading("h2"), h3: heading("h3"), h4: heading("h4"), h5: heading("h5"), h6: heading("h6"),
    // A link is shown as its text, followed by the destination as plain text when it is http(s)/mailto.
    a: ({ href, children }) => {
      stats.links++;
      const shown = displayUrl(href);
      if (!shown) stats.linksWithoutDisplayableUrl++;
      return h("span", { className: "md-link" }, children, shown ? ` (${shown})` : null);
    },
    // An image is never requested. Its alt text is shown instead.
    img: ({ alt }) => {
      stats.images++;
      return h("span", { className: "md-image-omitted" }, `[Image not shown${alt ? `: ${alt}` : ""}]`);
    },
  };
}

export function renderGuideMarkdown(markdown) {
  const stats = { links: 0, linksWithoutDisplayableUrl: 0, images: 0, treeDepth: 0, plainTextFallback: false };
  const html = renderToStaticMarkup(
    h(Markdown, {
      skipHtml: false, // default: raw HTML is shown as escaped text, never parsed (no rehype-raw)
      allowedElements: ALLOWED_ELEMENTS,
      unwrapDisallowed: true,
      urlTransform: displayUrl,
      components: guideComponents(stats),
      remarkPlugins: [remarkDepthGuard(stats, markdown)],
      rehypePlugins: [],
    }, markdown),
  );
  return { html, stats };
}

/** The current application renderer for comparison: React text in <p class="ai-copy"> (pre-wrap). */
export function renderCurrentPlainText(text) {
  return renderToStaticMarkup(h("p", { className: "small ai-copy" }, text));
}
