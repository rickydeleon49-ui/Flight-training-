// Generated boundary, encoding and privacy cases for the prototype validator.
// Built in memory so that multi-megabyte files and credential-shaped strings are never committed.
// Each case declares its expected outcome; run-fixtures.mjs compares it with the real validator result.
import { LIMITS } from "./validator.mjs";

const enc = (s) => new TextEncoder().encode(s);
const base = (extra = {}) => ({ format: "hfd-study-packet", formatVersion: 1, title: "Boundary", ...extra });
const guide = { summary: "x" };
const q = (i, extra = {}) => ({
  id: `q${i}`, type: "single-choice", question: `Pick label ${i}`,
  choices: [{ id: "a", text: "Zorb" }, { id: "b", text: "Quill" }], correctChoiceId: "b",
  explanation: "Invented.", ...extra,
});
const card = (i, extra = {}) => ({ id: `c${i}`, front: `Front ${i}`, back: `Back ${i}`, ...extra });
const json = (o) => enc(JSON.stringify(o));
const ok = (extra = {}) => ({ ok: true, ...extra });
const no = (stage, ...codes) => ({ ok: false, stage, codes: [...new Set(codes)].sort() });

export const SENTINEL = "SENTINEL-PRIVATE-7f3a";

function countValues(v) {
  if (Array.isArray(v)) return 1 + v.reduce((n, x) => n + countValues(x), 0);
  if (v && typeof v === "object") return 1 + Object.values(v).reduce((n, x) => n + countValues(x), 0);
  return 1;
}

function padTo(bytes, target) {
  const out = new Uint8Array(target);
  out.set(bytes);
  out.fill(0x20, bytes.length);
  return out;
}

function nestArrays(levels) {
  let v = 0;
  for (let k = 0; k < levels; k++) v = [v];
  return v;
}

function nestObjects(levels) {
  let v = 0;
  for (let k = 0; k < levels; k++) v = { a: v };
  return v;
}

export function boundaryCases() {
  const cases = [];
  const add = (name, bytes, expected, note) => cases.push({ name, bytes, expected, note });

  // Encoding and bytes (stage 1).
  add("empty-file", new Uint8Array(0), no("bytes", "file_empty"));
  add("utf16le-bom", new Uint8Array([0xff, 0xfe, 0x7b, 0x00, 0x7d, 0x00]), no("bytes", "unsupported_encoding"));
  add("invalid-utf8-sequence", new Uint8Array([...enc('{"title":"'), 0xc3, 0x28, ...enc('"}')]), no("bytes", "invalid_utf8"));
  add("overlong-utf8", new Uint8Array([...enc('{"title":"'), 0xc0, 0xaf, ...enc('"}')]), no("bytes", "invalid_utf8"));
  add("utf8-bom-accepted", new Uint8Array([0xef, 0xbb, 0xbf, ...json(base({ guide }))]), ok({ warnings: ["no_practice_test", "utf8_bom_removed"] }));

  // File size: 4 MiB reused text limit, padded with JSON whitespace.
  const small = json(base({ guide }));
  add("file-bytes-limit-minus-1", padTo(small, LIMITS.fileBytes - 1), ok());
  add("file-bytes-at-limit", padTo(small, LIMITS.fileBytes), ok());
  add("file-bytes-limit-plus-1", padTo(small, LIMITS.fileBytes + 1), no("bytes", "file_too_large"));

  // Depth (root object is depth 1). At the limit the scan passes and the schema rejects the unknown field.
  add("depth-at-limit-8", json(base({ guide, x: nestArrays(LIMITS.depth - 1) })), no("schema", "unknown_field"));
  add("depth-over-limit-9", json(base({ guide, x: nestArrays(LIMITS.depth) })), no("scan", "depth_exceeded"));
  add("depth-at-limit-8-objects", json(base({ guide, x: nestObjects(LIMITS.depth - 1) })), no("schema", "unknown_field"));
  add("depth-over-limit-9-objects", json(base({ guide, x: nestObjects(LIMITS.depth) })), no("scan", "depth_exceeded"));

  // Keys per object and key length.
  const keys = (n) => Object.fromEntries(Array.from({ length: n }, (_, k) => [`k${k}`, 0]));
  add("object-keys-at-limit-32", json({ ...base({ guide }), ...keys(LIMITS.objectKeys - 4) }), no("schema", "unknown_field"));
  add("object-keys-over-limit-33", json({ ...base({ guide }), ...keys(LIMITS.objectKeys - 3) }), no("scan", "too_many_keys"));
  add("key-length-at-limit-64", json(base({ guide, ["k".repeat(64)]: 0 })), no("schema", "unknown_field"));
  add("key-length-over-limit-65", json(base({ guide, ["k".repeat(65)]: 0 })), no("scan", "key_too_long"));

  // Integer digits.
  add("integer-9-digits", json(base({ flashcards: [card(1, { sourceRefs: [{ page: 123456789 }] })] })), no("schema", "page_out_of_range"));
  add("integer-10-digits", json(base({ flashcards: [card(1, { sourceRefs: [{ page: 1234567890 }] })] })), no("scan", "unsupported_number"));
  add("page-at-max-1000", json(base({ flashcards: [card(1, { sourceRefs: [{ page: 1000 }] })] })), ok({ warnings: ["no_practice_test", "source_refs_unverified"] }));

  // Total values.
  const valuesBase = base({ guide, x: [] });
  const room = LIMITS.values - countValues(valuesBase);
  add("values-at-limit", json(base({ guide, x: new Array(room).fill(0) })), no("schema", "unknown_field"));
  add("values-over-limit", json(base({ guide, x: new Array(room + 1).fill(0) })), no("scan", "too_many_values"));

  // String limits are Unicode code points: 200 astral emoji are 400 UTF-16 units but 200 code points.
  add("title-199-code-points", json(base({ guide, title: "\u{1F680}".repeat(199) })), ok());
  add("title-200-code-points-astral", json(base({ guide, title: "\u{1F680}".repeat(200) })), ok());
  add("title-201-code-points", json(base({ guide, title: "\u{1F680}".repeat(201) })), no("schema", "too_long"));
  add("title-nfc-200-after-normalization", json(base({ guide, title: "é".repeat(200) })), ok(), "400 code points before NFC, 200 after");
  add("body-20000", json(base({ guide: { sections: [{ heading: "h", body: "b".repeat(20_000) }] } })), ok());
  add("body-20001", json(base({ guide: { sections: [{ heading: "h", body: "b".repeat(20_001) }] } })), no("schema", "too_long"));

  // Array limits.
  const choices = (n) => Array.from({ length: n }, (_, k) => ({ id: `c${k}`, text: `Label ${k}` }));
  add("choices-6", json(base({ questions: [q(1, { choices: choices(6), correctChoiceId: "c0" })] })), ok());
  add("choices-7", json(base({ questions: [q(1, { choices: choices(7), correctChoiceId: "c0" })] })), no("schema", "too_many_items"));
  add("choices-2", json(base({ questions: [q(1)] })), ok());
  add("questions-200", json(base({ questions: Array.from({ length: 200 }, (_, k) => q(k)) })), ok());
  add("questions-201", json(base({ questions: Array.from({ length: 201 }, (_, k) => q(k)) })), no("schema", "too_many_items"));
  add("flashcards-500", json(base({ flashcards: Array.from({ length: 500 }, (_, k) => card(k)) })), ok());
  add("flashcards-501", json(base({ flashcards: Array.from({ length: 501 }, (_, k) => card(k)) })), no("schema", "too_many_items"));
  const sections = (n) => Array.from({ length: n }, (_, k) => ({ heading: `H${k}`, body: "b" }));
  add("sections-100", json(base({ guide: { sections: sections(100) } })), ok());
  add("sections-101", json(base({ guide: { sections: sections(101) } })), no("schema", "too_many_items"));
  const refs = (n) => Array.from({ length: n }, (_, k) => ({ label: `Ref ${k}` }));
  add("source-refs-5", json(base({ flashcards: [card(1, { sourceRefs: refs(5) })] })), ok());
  add("source-refs-6", json(base({ flashcards: [card(1, { sourceRefs: refs(6) })] })), no("schema", "too_many_items"));

  // Identifier lengths.
  add("id-64", json(base({ flashcards: [card(1, { id: "a".repeat(64) })] })), ok());
  add("id-65", json(base({ flashcards: [card(1, { id: "a".repeat(65) })] })), no("schema", "invalid_id"));
  add("choice-id-32", json(base({ questions: [q(1, { choices: [{ id: "x".repeat(32), text: "A" }, { id: "b", text: "B" }], correctChoiceId: "b" })] })), ok());
  add("choice-id-33", json(base({ questions: [q(1, { choices: [{ id: "x".repeat(33), text: "A" }, { id: "b", text: "B" }], correctChoiceId: "b" })] })), no("schema", "invalid_id"));

  // Credential-like content (assembled here so no credential-shaped literal is committed).
  const googleLike = "AIza" + "B".repeat(35);
  const pemLike = "-----BEGIN " + "PRIVATE KEY-----";
  add("credential-google-key-shape", json(base({ guide: { summary: `Key: ${googleLike} ${SENTINEL}` } })), no("semantic", "credential_like_content"));
  add("credential-pem-header", json(base({ flashcards: [card(1, { back: `${pemLike}\nabc` })] })), no("semantic", "credential_like_content"));

  // Error cap: 60 semantic errors -> 50 reported, 10 omitted.
  add("error-cap-60", json(base({ questions: Array.from({ length: 60 }, (_, k) => q(k, { correctChoiceId: "z" })) })), { ...no("semantic", "answer_key_not_in_choices"), reported: 50, omitted: 10 });

  // Privacy sentinels in rejected packets: the sentinel must never appear in errors.
  add("sentinel-wrong-type", json(base({ guide: { summary: SENTINEL, sections: SENTINEL } })), no("schema", "wrong_type"));
  add("sentinel-bad-answer", json(base({ questions: [q(1, { question: SENTINEL, correctChoiceId: "zz" })] })), no("semantic", "answer_key_not_in_choices"));
  add("sentinel-unknown-field-name", json(base({ guide, [`${SENTINEL} with spaces`]: SENTINEL })), no("schema", "unknown_field"));
  add("sentinel-duplicate-key-unsafe-name", enc(`{"format":"hfd-study-packet","formatVersion":1,"title":"x","guide":{"summary":"x"},"${SENTINEL} !":1,"${SENTINEL} !":2}`), no("scan", "duplicate_key"));
  return cases;
}
