function err(code, path, extra = {}) {
  return { code, path, message: MESSAGES[code] ?? MESSAGES.invalid_value, ...extra };
}

class ScanError extends Error {
  constructor(issue) {
    super(issue.code);
    this.issue = issue;
  }
}

function pathString(parts) {
  let out = "$";
  for (const part of parts) {
    if (typeof part === "number") out += `[${part}]`;
    else if (CONTRACT_FIELDS.has(part)) out += `.${part}`;
    else out += ".<field>";
  }
  return out;
}

const PUBLIC_NUMBER_KEYS = ["limit", "offset", "index", "firstIndex"];
const FIELD_ALLOWLIST = new Set([...CONTRACT_FIELDS, ...DIRECTIVE_FIELDS, ...LEGACY_ANSWER_FIELDS]);

/**
 * The diagnostic boundary. Every error leaving the validator is rebuilt from fixed parts only: a known code and
 * its fixed message, a path made of contract-owned names and integer indexes, integer limits/offsets/indexes,
 * the fixed supported-version list, and `field` only when it is a contract-owned or allowlisted name.
 */
function publicError(e) {
  const code = Object.prototype.hasOwnProperty.call(MESSAGES, e?.code) ? e.code : "internal_error";
  const path = typeof e?.path === "string"
    ? e.path.replace(/\.([^.\[]+)/g, (m, name) => (CONTRACT_FIELDS.has(name) || name === "<field>" ? m : ".<field>"))
    : "$";
  const out = { code, path: /^\$(?:\[\d+\]|\.(?:<field>|[A-Za-z]+))*$/.test(path) ? path : "$", message: MESSAGES[code] };
  for (const k of PUBLIC_NUMBER_KEYS) if (Number.isInteger(e?.[k])) out[k] = e[k];
  if (code === "unsupported_version") out.supported = [FORMAT_VERSION];
  if (typeof e?.field === "string" && FIELD_ALLOWLIST.has(e.field)) out.field = e.field;
  return out;
}

function failure(stage, errors) {
  const capped = errors.length > LIMITS.reportedErrors ? errors.slice(0, LIMITS.reportedErrors) : errors;
  return { ok: false, stage, errors: capped.map(publicError), omittedErrors: Math.max(0, errors.length - LIMITS.reportedErrors) };
}

function codePointLength(s) {
  let n = 0;
  for (const _ of s) n++;
  return n;
}

// Rejects C0/C1 controls (except tab, LF, CR), DEL and lone surrogates. Written as a loop because
// String.prototype.isWellFormed and regex lookbehind are unavailable on macOS 12.0 WebKit.
function hasInvalidCharacters(s) {
  for (let i = 0; i < s.length; i++) {
    const c = s.charCodeAt(i);
    if (c >= 0xd800 && c <= 0xdbff) {
      const next = s.charCodeAt(i + 1);
      if (next >= 0xdc00 && next <= 0xdfff) {
        i++;
        continue;
      }
      return true;
    }
    if (c >= 0xdc00 && c <= 0xdfff) return true;
    if (c === 0x09 || c === 0x0a || c === 0x0d) continue;
    if (c < 0x20 || (c >= 0x7f && c <= 0x9f)) return true;
  }
  return false;
}

/** Stage 2: strict RFC 8259 scanner with bounds. Builds the value itself; JSON.parse is never used. */
export function scanJson(text) {
  let i = 0;
  let values = 0;
  const path = [];

  const fail = (code, extra = {}) => {
    throw new ScanError(err(code, pathString(path), { offset: i, ...extra }));
  };
  const ws = () => {
    while (i < text.length) {
      const c = text.charCodeAt(i);
      if (c === 0x20 || c === 0x09 || c === 0x0a || c === 0x0d) i++;
      else break;
    }
  };
  const countValue = () => {
    if (++values > LIMITS.values) fail("too_many_values");
  };

  const parseString = () => {
    // text[i] === '"'
    i++;
    let out = "";
    let start = i;
    for (;;) {
      if (i >= text.length) fail("malformed_json");
      const c = text.charCodeAt(i);
      if (c === 0x22) {
        out += text.slice(start, i);
        i++;
        break;
      }
      if (c < 0x20) fail("malformed_json"); // raw control characters are not JSON
      if (c === 0x5c) {
        out += text.slice(start, i);
        const e = text[i + 1];
        const simple = { '"': '"', "\\": "\\", "/": "/", b: "\b", f: "\f", n: "\n", r: "\r", t: "\t" }[e];
        if (simple !== undefined) {
          out += simple;
          i += 2;
        } else if (e === "u") {
          const hex = text.slice(i + 2, i + 6);
          if (!/^[0-9A-Fa-f]{4}$/.test(hex)) fail("malformed_json");
          out += String.fromCharCode(parseInt(hex, 16));
          i += 6;
        } else {
          fail("malformed_json");
        }
        start = i;
        continue;
      }
      i++;
    }
    if (hasInvalidCharacters(out)) fail("invalid_string_character");
    return out;
  };

  const parseNumber = () => {
    const m = /^-?(0|[1-9][0-9]*)(\.[0-9]+)?([eE][+-]?[0-9]+)?/.exec(text.slice(i, i + 64));
    if (!m) fail("malformed_json");
    if (m[2] || m[3] || m[1].length > LIMITS.integerDigits) fail("unsupported_number");
    i += m[0].length;
    return Number(m[0]);
  };

  const parseValue = (depth) => {
    ws();
    if (i >= text.length) fail("malformed_json");
    countValue();
    const ch = text[i];
    if (ch === "{") {
      if (depth + 1 > LIMITS.depth) fail("depth_exceeded", { limit: LIMITS.depth });
      i++;
      const obj = {};
      const seen = new Set();
      ws();
      if (text[i] === "}") {
        i++;
        return obj;
      }
      for (;;) {
        ws();
        if (text[i] !== '"') fail("malformed_json");
        const key = parseString();
        if (FORBIDDEN_KEYS.has(key)) fail("forbidden_property_name");
        if (codePointLength(key) > LIMITS.keyCodePoints) fail("key_too_long", { limit: LIMITS.keyCodePoints });
        if (seen.has(key)) fail("duplicate_key", CONTRACT_FIELDS.has(key) ? { field: key } : {});
        seen.add(key);
        if (seen.size > LIMITS.objectKeys) fail("too_many_keys", { limit: LIMITS.objectKeys });
        ws();
        if (text[i] !== ":") fail("malformed_json");
        i++;
        path.push(key);
        obj[key] = parseValue(depth + 1);
        path.pop();
        ws();
        if (text[i] === ",") {
          i++;
          continue;
        }
        if (text[i] === "}") {
          i++;
          return obj;
        }
        fail("malformed_json");
      }
    }
    if (ch === "[") {
      if (depth + 1 > LIMITS.depth) fail("depth_exceeded", { limit: LIMITS.depth });
      i++;
      const arr = [];
      ws();
      if (text[i] === "]") {
        i++;
        return arr;
      }
      for (;;) {
        path.push(arr.length);
        arr.push(parseValue(depth + 1));
        path.pop();
        ws();
        if (text[i] === ",") {
          i++;
          continue;
        }
        if (text[i] === "]") {
          i++;
          return arr;
        }
        fail("malformed_json");
      }
    }
    if (ch === '"') return parseString();
    if (ch === "-" || (ch >= "0" && ch <= "9")) return parseNumber();
    if (text.startsWith("true", i)) {
      i += 4;
      return true;
    }
    if (text.startsWith("false", i)) {
      i += 5;
      return false;
    }
    if (text.startsWith("null", i)) {
      i += 4;
      return null;
    }
    fail("malformed_json");
  };

  try {
    const value = parseValue(0);
    ws();
    if (i !== text.length) fail("trailing_content");
    return { ok: true, value };
  } catch (e) {
    if (e instanceof ScanError) return { ok: false, error: e.issue };
    throw e;
  }
}

/** Stage 1: bytes to text. */
export function decodeBytes(bytes) {
  if (bytes.length === 0) return { ok: false, error: err("file_empty", "$") };
  if (bytes.length > LIMITS.fileBytes) return { ok: false, error: err("file_too_large", "$", { limit: LIMITS.fileBytes }) };
  if ((bytes[0] === 0xfe && bytes[1] === 0xff) || (bytes[0] === 0xff && bytes[1] === 0xfe)) {
    return { ok: false, error: err("unsupported_encoding", "$") };
  }
  const warnings = [];
  let view = bytes;
  if (bytes[0] === 0xef && bytes[1] === 0xbb && bytes[2] === 0xbf) {
    view = bytes.subarray(3);
    warnings.push("utf8_bom_removed");
  }
  let text;
  try {
    text = new TextDecoder("utf-8", { fatal: true, ignoreBOM: true }).decode(view);
  } catch {
    return { ok: false, error: err("invalid_utf8", "$") };
  }
  if (text.length === 0) return { ok: false, error: err("file_empty", "$") };
  return { ok: true, text, warnings };
}

// Deterministic normalization (contract §2.7).
export function normLine(s) {
  return s.replace(/\r\n?/g, "\n").normalize("NFC").replace(/\s+/g, " ").trim();
}
export function normBlock(s) {
  return s.replace(/\r\n?/g, "\n").normalize("NFC").replace(/^(?:[ \t]*\n)+/, "").replace(/\s+$/, "");
}

function text(kind, { min = 0, max }) {
  return z.string().transform((raw, ctx) => {
    const value = kind === "line" ? normLine(raw) : normBlock(raw);
    const length = codePointLength(value);
    if (min > 0 && length === 0) {
      ctx.issues.push({ code: "custom", input: raw, params: { hfd: "empty_required" } });
      return z.NEVER;
    }
    if (length > max) {
      ctx.issues.push({ code: "custom", input: raw, params: { hfd: "too_long", limit: max } });
      return z.NEVER;
    }
    return value;
  });
}

function lastKey(path) {
  for (let k = path.length - 1; k >= 0; k--) if (typeof path[k] === "string") return path[k];
  return undefined;
}

/** Map Zod issues to contract error codes. Never copies `input` into the result. */
function mapIssue(issue) {
  const path = pathString(issue.path);
  if (issue.code === "custom" && issue.params?.hfd) {
    return err(issue.params.hfd, path, issue.params.limit ? { limit: issue.params.limit } : {});
  }
  if (issue.code === "unrecognized_keys") {
    return issue.keys.map((key) => {
      if (DIRECTIVE_FIELDS.has(key)) return err("directive_field_unsupported", path, { field: key });
      if (LEGACY_ANSWER_FIELDS.has(key)) return err("ambiguous_answer_key_field", path, { field: key });
      return err("unknown_field", path); // the unknown name itself is never returned
    });
  }
  const key = lastKey(issue.path);
  if (issue.code === "invalid_type") {
    const input = issue.input;
    if (input === undefined) return err("required_missing", path);
    if (key === "correctChoiceId" && typeof input === "number") return err("answer_key_type", path);
    if (key === "correctChoiceId" && Array.isArray(input)) return err("multiple_answers_unsupported", path);
    if (input === null) return err("null_not_allowed", path);
    if (key === "page" && typeof input === "number") return err("page_out_of_range", path);
    return err("wrong_type", path);
  }
  if (issue.code === "invalid_value") {
    if (key === "type") return err("unsupported_question_type", path);
    return err("invalid_value", path);
  }
  if (issue.code === "invalid_format") return err("invalid_id", path);
  if (issue.code === "too_big") {
    if (key === "page") return err("page_out_of_range", path);
    if (issue.origin === "array") return err("too_many_items", path, { limit: Number(issue.maximum) });
    return err("too_long", path);
  }
  if (issue.code === "too_small") {
    if (key === "page") return err("page_out_of_range", path);
    if (issue.origin === "array") return err("too_few_items", path, { limit: Number(issue.minimum) });
    return err("empty_required", path);
  }
  return err("invalid_value", path);
}

/** Stage 5: identity, answer keys, choices and credential-like content on the normalized value. */
export function semanticCheck(packet) {
  const errors = [];
  const warnings = new Set();

  const hasGuide = Boolean(packet.guide && (packet.guide.summary || (packet.guide.sections ?? []).length));
  const cards = packet.flashcards ?? [];
  const questions = packet.questions ?? [];
  if (!hasGuide && cards.length === 0 && questions.length === 0) errors.push(err("packet_empty", "$"));

  const uniq = (items, base) => {
    const seen = new Map();
    items.forEach((item, index) => {
      const k = item.id.toLowerCase();
      if (seen.has(k)) errors.push(err("duplicate_id", `${base}[${index}].id`, { index, firstIndex: seen.get(k) }));
      else seen.set(k, index);
    });
  };
  uniq(cards, "$.flashcards");
  uniq(questions, "$.questions");

  questions.forEach((q, qi) => {
    const base = `$.questions[${qi}]`;
    const ids = new Map();
    const texts = new Map();
    q.choices.forEach((c, ci) => {
      const k = c.id.toLowerCase();
      if (ids.has(k)) errors.push(err("duplicate_choice_id", `${base}.choices[${ci}].id`, { index: ci, firstIndex: ids.get(k) }));
      else ids.set(k, ci);
      const t = c.text.toLowerCase();
      if (texts.has(t)) errors.push(err("duplicate_choice_text", `${base}.choices[${ci}].text`, { index: ci, firstIndex: texts.get(t) }));
      else texts.set(t, ci);
    });
    const exact = q.choices.some((c) => c.id === q.correctChoiceId);
    if (!exact) {
      const folded = ids.has(q.correctChoiceId.toLowerCase());
      errors.push(err(folded ? "answer_key_case_mismatch" : "answer_key_not_in_choices", `${base}.correctChoiceId`));
    }
  });

  const refsSeen = [];
  const checkRefs = (refs, base) => {
    (refs ?? []).forEach((r, ri) => {
      if (r.label === undefined && r.page === undefined) errors.push(err("source_ref_empty", `${base}.sourceRefs[${ri}]`));
      else if (r.label === "" && r.page === undefined) errors.push(err("source_ref_empty", `${base}.sourceRefs[${ri}]`));
      refsSeen.push(r);
    });
  };
  cards.forEach((c, i) => checkRefs(c.sourceRefs, `$.flashcards[${i}]`));
  questions.forEach((q, i) => checkRefs(q.sourceRefs, `$.questions[${i}]`));
  if (refsSeen.length) warnings.add("source_refs_unverified");
  if (sourceClaims(packet.source)) warnings.add("source_claim_unverified");

  // Walk every normalized string with its path.
  const walk = (value, p) => {
    if (typeof value === "string") {
      if (CREDENTIAL_PATTERNS.some((re) => re.test(value))) errors.push(err("credential_like_content", p));
      if (BIDI_RE.test(value)) warnings.add("bidi_controls_present");
      if (HTML_LIKE_RE.test(value)) warnings.add("html_like_text");
      if (MD_IMAGE_RE.test(value)) warnings.add("markdown_images_not_shown");
    } else if (Array.isArray(value)) {
      value.forEach((v, k) => walk(v, `${p}[${k}]`));
    } else if (value && typeof value === "object") {
      for (const [k, v] of Object.entries(value)) walk(v, `${p}.${k}`);
    }
  };
  walk(packet, "$");

  if (questions.length === 0 && (hasGuide || cards.length)) warnings.add("no_practice_test");
  if (questions.length > 0 && !hasGuide && cards.length === 0) warnings.add("no_study_guide");
  return { errors, warnings: [...warnings], hasGuide };
}

/** Non-empty normalized source claims, or null. An empty object or blank values are not a claim. */
function sourceClaims(source) {
  if (!source) return null;
  const claims = {};
  for (const k of ["generator", "createdAt", "note"]) if (source[k]) claims[k] = source[k];
  return Object.keys(claims).length ? claims : null;
}

const refsFor = (refs) => (refs ?? []).map((r) => ({ ...(r.label ? { label: r.label } : {}), ...(r.page !== undefined ? { page: r.page } : {}) }));

/**
 * Bounded, normalized, in-memory preview data (contract §2.13, PacketPreviewData). Built from the same validated
 * value as the persistence payload; the original file is never parsed again. Holds unverified source claims, export
 * IDs and item-associated references for display; none of these enter the persistence payload.
 */
export function toPreviewData(packet, hasGuide) {
  const cards = packet.flashcards ?? [];
  const questions = packet.questions ?? [];
  const source = sourceClaims(packet.source);
  const data = {
    title: packet.title,
    source,
    guide: hasGuide || packet.guide?.sourceNotes
      ? { summary: packet.guide?.summary ?? "", sections: (packet.guide?.sections ?? []).map((s) => ({ heading: s.heading, body: s.body })), sourceNotes: packet.guide?.sourceNotes ?? "" }
      : null,
    flashcards: cards.map((c, index) => ({ index, id: c.id, front: c.front, back: c.back, sourceRefs: refsFor(c.sourceRefs) })),
    questions: questions.map((q, index) => ({
      index, id: q.id, question: q.question, choices: q.choices.map((c) => ({ id: c.id, text: c.text })),
      correctChoiceId: q.correctChoiceId, explanation: q.explanation, sourceRefs: refsFor(q.sourceRefs),
    })),
  };
  const discardedOnSave = {
    sourceClaims: source ? Object.keys(source) : [],
    sourceRefs: data.flashcards.reduce((n, c) => n + c.sourceRefs.length, 0) + data.questions.reduce((n, q) => n + q.sourceRefs.length, 0),
    exportIds: cards.length + questions.length,
    choiceIds: questions.reduce((n, q) => n + q.choices.length, 0),
  };
  return { data, discardedOnSave };
}

/**
 * Material defaults are NOT part of the payload. They are fixed constants owned by the future native save
 * command (Slice B, save_structured_study_import), so a packet cannot set them. Listed here for the fake adapter.
 */
export const IMPORT_MATERIAL_DEFAULTS = Object.freeze({
  materialType: "study_guide", topic: "general", reviewStatus: "needs_review", priority: "later",
  goalId: null, eventId: null, notes: "", filePath: null, originalFileName: null, fileSize: null, lastReviewedAt: null, tags: [],
});

/**
 * Canonical persistence JSON for the content hash (contract §3.5): fixed field order, persistence fields only.
 * Excludes every preview-only value (source claims, export IDs, references, warnings, counts) and the original
 * file bytes. contentHash = SHA-256 of this string's UTF-8 bytes, computed by the caller.
 */
export function canonicalPersistenceJson(payload) {
  const set = payload.studySet;
  const test = payload.practiceTest;
  return JSON.stringify([
    "hfd-study-packet-content/1",
    [payload.material.title],
    set ? [set.provider, set.model, set.title, set.summary, set.sections.map((s) => [s.heading, s.body]),
      set.flashcards.map((c) => [c.front, c.back, c.sourcePage]), set.sourceNotes] : null,
    test ? [test.provider, test.model, test.title,
      test.questions.map((x) => [x.question, x.options, x.correctIndex, x.explanation, x.sourcePage])] : null,
  ]);
}

/** Map a validated packet to the import persistence DTOs (contract §2.13; provider "import", not the stored domain type). */
export function toApplicationPayload(packet, hasGuide) {
  const cards = packet.flashcards ?? [];
  const questions = packet.questions ?? [];
  const studySet =
    hasGuide || cards.length
      ? {
          title: packet.title,
          summary: packet.guide?.summary ?? "",
          sections: (packet.guide?.sections ?? []).map((s) => ({ heading: s.heading, body: s.body })),
          flashcards: cards.map((c) => ({ front: c.front, back: c.back, sourcePage: null })),
          sourceNotes: packet.guide?.sourceNotes ?? "",
          provider: "import",
          model: "",
        }
      : null;
  const practiceTest = questions.length
    ? {
        title: packet.title,
        questions: questions.map((q) => ({
          question: q.question,
          options: q.choices.map((c) => c.text),
          correctIndex: q.choices.findIndex((c) => c.id === q.correctChoiceId),
          explanation: q.explanation,
          sourcePage: null,
        })),
        provider: "import",
        model: "",
      }
    : null;
  return { material: { title: packet.title }, studySet, practiceTest };
}

/**
 * Pipeline from original bytes (tests, browser adapter). The native reader (proposed pick_study_packet) performs the
 * same byte checks on the original file and passes validated text to validatePacketText.
 */
export function validatePacketBytes(bytes) {
  try {
    if (!(bytes instanceof Uint8Array)) return failure("internal", [{ code: "internal_error", path: "$" }]);
    const decoded = decodeBytes(bytes);
    if (!decoded.ok) return failure("bytes", [decoded.error]);
    return validateDecoded(decoded.text, decoded.warnings);
  } catch {
    return failure("internal", [{ code: "internal_error", path: "$" }]);
  }
}

/**
 * Pipeline from text already decoded by the native reader. The byte limit is re-derived from the text itself
 * (UTF-8 length, plus 3 when the reader removed a BOM); a caller-claimed length is never trusted.
 */
export function validatePacketText(text, { bomRemoved = false } = {}) {
  try {
    if (typeof text !== "string") return failure("internal", [{ code: "internal_error", path: "$" }]);
    const bytes = new TextEncoder().encode(text).length + (bomRemoved ? 3 : 0);
    if (bytes === 0 || text.length === 0) return failure("bytes", [err("file_empty", "$")]);
    if (bytes > LIMITS.fileBytes) return failure("bytes", [err("file_too_large", "$", { limit: LIMITS.fileBytes })]);
    if (text.charCodeAt(0) === 0xfeff) return failure("bytes", [err("unsupported_encoding", "$")]);
    return validateDecoded(text, bomRemoved ? ["utf8_bom_removed"] : []);
  } catch {
    return failure("internal", [{ code: "internal_error", path: "$" }]);
  }
}

function validateDecoded(text, byteWarnings) {
  const scanned = scanJson(text);
  if (!scanned.ok) return failure("scan", [scanned.error]);
  const root = scanned.value;
  if (!root || typeof root !== "object" || Array.isArray(root)) return failure("gate", [err("root_not_object", "$")]);
  if (!("format" in root)) return failure("gate", [err("format_missing", "$.format")]);
  if (root.format !== FORMAT) return failure("gate", [err("unsupported_format", "$.format")]);
  if (!("formatVersion" in root)) return failure("gate", [err("version_missing", "$.formatVersion")]);
  if (root.formatVersion !== FORMAT_VERSION) return failure("gate", [err("unsupported_version", "$.formatVersion")]);
  const parsed = packetSchema.safeParse(root, { reportInput: true });
  if (!parsed.success) return failure("schema", parsed.error.issues.flatMap((issue) => mapIssue(issue)));
  const semantic = semanticCheck(parsed.data);
  if (semantic.errors.length) return failure("semantic", semantic.errors);
  const payload = toApplicationPayload(parsed.data, semantic.hasGuide);
  const { data, discardedOnSave } = toPreviewData(parsed.data, semantic.hasGuide);
  const warnings = [...byteWarnings, ...semantic.warnings];
  return {
    ok: true,
    stage: "preview",
    preview: {
      title: parsed.data.title,
      counts: {
        sections: data.guide?.sections.length ?? 0,
        flashcards: data.flashcards.length,
        questions: data.questions.length,
        choices: discardedOnSave.choiceIds,
        sourceRefs: discardedOnSave.sourceRefs,
      },
      createsStudySet: payload.studySet !== null,
      createsPracticeTest: payload.practiceTest !== null,
      warnings: warnings.map((code) => ({ code, message: WARNING_MESSAGES[code] })),
      data,
      discardedOnSave,
    },
    payload,
  };
}
