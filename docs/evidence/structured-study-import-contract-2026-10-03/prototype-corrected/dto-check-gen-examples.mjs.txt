// Writes examples.generated.ts from the ACTUAL corrected prototype output for guide-only (01), test-only (04),
// combined (02) and references (57) fixtures, typed against the DTOs. Usage: node gen-examples.mjs <proto> <fixtures>
import { readFileSync, writeFileSync } from "node:fs";
import { join } from "node:path";
const [proto, fixtures] = process.argv.slice(2);
const { validatePacketBytes } = await import(join(proto, "src/validator.mjs"));
const pick = (f) => { const r = validatePacketBytes(readFileSync(join(fixtures, f))); if (!r.ok) throw new Error(f); return r; };
const cases = { guideOnly: "01-guide-only.json", testOnly: "04-questions-only.json", combined: "02-flashcards-and-test.json", withRefs: "57-refs-on-multiple-items.json" };
let out = '// Generated from the corrected prototype output. Do not edit.\nimport type { ImportPayload, PacketValidationResult } from "./import-dto";\n\n';
for (const [name, file] of Object.entries(cases)) {
  const r = pick(file);
  out += `export const ${name}Payload: ImportPayload = ${JSON.stringify(r.payload, null, 2)};\n\n`;
  out += `export const ${name}Result: PacketValidationResult = ${JSON.stringify(r, null, 2)};\n\n`;
}
const fail = validatePacketBytes(Buffer.from('{"format":"hfd-study-packet","formatVersion":1,"title":"x","guide":{"summary":"x"},"actions":[1],"zz":1}'));
out += `export const failureResult: PacketValidationResult = ${JSON.stringify(fail, null, 2)};\n`;
writeFileSync("examples.generated.ts", out);
console.log("written", Object.keys(cases).length * 2 + 1, "typed values");
