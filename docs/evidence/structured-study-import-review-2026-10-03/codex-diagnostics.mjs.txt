import {validatePacketBytes} from './src/validator.mjs';
import {readFileSync,writeFileSync} from 'node:fs';
const fixtureDir=process.argv[2];
if(!fixtureDir) throw new Error('Usage: node codex-diagnostics.mjs <fixtures-dir>');
const base=JSON.parse(readFileSync(new URL('file://'+fixtureDir+'/02-flashcards-and-test.json'),'utf8'));
const run=p=>validatePacketBytes(Buffer.from(JSON.stringify(p)));
const clone=()=>structuredClone(base);
const rows=[];
function check(name, expected, actual){rows.push({name,expected,actual,pass:JSON.stringify(expected)===JSON.stringify(actual)});}
const secret='sk-'+ 'Q'.repeat(30);
let p=clone();p[secret]='ordinary';let r=run(p);
check('R1: unknown identifier-shaped key never echoed',false,JSON.stringify(r.errors).includes(secret));
p=clone();p.questions[0].id=secret;p.questions[1].id=secret;r=run(p);
check('R1: duplicate credential-shaped item id never echoed',false,JSON.stringify(r.errors).includes(secret));
let text='{"'+secret+'":{"title":"x", "title":"y"}}';r=validatePacketBytes(Buffer.from(text));
check('R1: scan-stage path never echoes unknown identifier',false,JSON.stringify(r.errors).includes(secret));
r=run(clone());
check('R2: validated preview retains normalized source claims',true,Boolean(r.packet?.source || r.preview.packet?.source || r.preview.source));
check('R2: validated preview retains per-item unverified references',true,Boolean(r.packet?.questions?.[0]?.sourceRefs || r.preview.packet?.questions?.[0]?.sourceRefs || r.preview.questions?.[0]?.sourceRefs || r.preview.sourceRefs));
p=clone();p.source={note:'Invented exporter claim',createdAt:'Invented date'};r=run(p);
check('R2: source note/date without generator are marked unverified',true,r.preview.warnings.some(w=>w.code==='source_claim_unverified'));
p=clone();p.flashcards[0].id='c1\n';r=run(p);
check('Control: trailing LF in flashcard id is rejected',false,r.ok);
p=clone();p.questions[0].choices[0].id='a\n';p.questions[0].correctChoiceId='a\n';r=run(p);
check('Control: trailing LF in choice id and answer is rejected',false,r.ok);
const summary={total:rows.length,passed:rows.filter(x=>x.pass).length,failed:rows.filter(x=>!x.pass).length,node:process.version};
writeFileSync('results/codex-diagnostics.json',JSON.stringify({summary,rows},null,2)+'\n');
console.log(JSON.stringify({summary,rows},null,2));
process.exit(summary.failed?1:0);
